596             path = out / f"{page:04d}.{fmt}"
597             path.write_bytes(body)
598             records.append(PageImageRecord(page=page, url=raw_url, scheme_used=scheme,
599                                            status=status, fmt=fmt, byte_len=len(body),
600                                            path=path))
601             if self._store is not None and doc:
602                 self._store.unfailed(site=SITE, doc=doc, page=page)
603             self._audit_row("page_image", doc, len(body), "ok",
604                             f"page={page} scheme={scheme} fmt={fmt} bytes={len(body)}")
605             if i < len(pages) - 1:
606                 self._sleep(PAGE_IMAGE_GAP_S)     # 页间礼貌间隔（可注入；SPEC 未钉图床频控）
607         self._audit_row("page_images", doc, len(records),
608                         "ok" if not failures else "fail",
609                         f"ok={len(records)} fail={len(failures)}"
610                         + ("; " + "; ".join(failures[:5]) if failures else ""))
611         if failures:
612             raise Book118ImageError(
613                 "page_image_failed",
614                 f"{len(failures)}/{len(pages)} 页失败: " + "; ".join(failures[:8]))
615         return records

617     def harvest_preview_pdf(self, meta: FileMeta, work_dir, *,
618                             image_transport: Callable[[str], tuple[int, bytes]] | None = None,
619                             ocr_fn: Callable | None = None) -> "PreviewPdfResult":
620         """FR3.1 收割腿全链：页清单→图直下→魔数识别→int 排序→像素拼 PDF→OCR 参照→质检门。
621 
622         前置=fetch_ladder 已把 book118_page_urls/页数入 meta.extra（T19 口径：int 页号→
623         协议相对 URL）；无页清单显式 Book118ImageError(no_page_list)。
624 
625         - 像素拼 PDF=pdf/image_pdf.build_image_pdf（页序=页号 int 序，页几何=图像素 1px=1pt），
626           页数 pypdfium2 实测（数字零造假）；
627         - 收费形态（preview<actual / book118_partial_preview）：FileMeta 形态标注
628           「预览部分」+perm=yellow（黄线透明），manifest 用 preview_limit（quality_gate
629           部分收割合法口径）；全预览形态 manifest 用 expected_pages（==actual）；
630         - 质检门参照=ocr_fn 产物（缺省 T13 make_pdf_searchable 全文 OCR 实测 chars，勿
631           手填；ref_chars_source=book118——免费件文字层 GT 取词协议归 T33）；门②字数吃
632           文字层 → 质检门目标 PDF=带 OCR 文字层的 searchable PDF（与图 PDF 同页数同页序）；
633         - ocr_fn 注入点④：Callable[[图PDF, 出PDF], int]（OCR 子系统边界，离线确定性）。
634         """
635         extra = meta.extra or {}
636         page_urls = extra.get("book118_page_urls") or {}
637         if not page_urls:
638             raise Book118ImageError(
639                 "no_page_list",
640                 f"meta.extra 无 book118_page_urls（先跑 fetch_ladder）：doc_id="
641                 f"{extra.get('book118_doc_id')!r}")
642         doc_id = str(extra.get("book118_doc_id") or "")
643         work = Path(work_dir)
644         records = self.download_page_images(page_urls, work / "pages", doc_id,
645                                             image_transport=image_transport)
646         build_image_pdf, _ = _import_image_stack()
647         stem = doc_id or "book118"
648         image_pdf = build_image_pdf([r.path for r in records], work / f"{stem}.pdf",
649                                     title=meta.title or None)
650         pdfium = _import_pdfium()
651         with pdfium.PdfDocument(str(image_pdf)) as doc:
652             pages_measured = len(doc)            # 实测页数（不拿清单长度冒充）
653         preview = extra.get("book118_pages_preview")
654         actual = extra.get("book118_pages_actual")
655         partial = bool(extra.get("book118_partial_preview")) or (
656             preview is not None and actual is not None and int(preview) < int(actual))
657         if partial:                              # 收费/部分预览形态：黄线+「预览部分」
658             meta.perm = "yellow"
659             meta.extra["book118_form"] = "预览部分"
660         searchable_pdf = work / f"{stem}_ocr.pdf"
661         ref_fn = ocr_fn or self._ocr_ref
662         ref_chars = int(ref_fn(image_pdf, searchable_pdf))
663         manifest = {"pdf_path": str(searchable_pdf), "ref_chars": ref_chars,
664                     "ref_chars_source": "book118"}
665         if partial:
666             manifest["preview_limit"] = int(preview if preview is not None else len(records))
667         else:
668             manifest["expected_pages"] = int(
669                 actual if actual is not None
670                 else (preview if preview is not None else len(records)))
671         manifest_path = work / f"{stem}_manifest.json"
672         manifest_path.write_text(json.dumps(manifest, ensure_ascii=False, indent=1),
673                                  encoding="utf-8")
674         quality_gate = _import_quality_gate()
675         checked = quality_gate.load_manifest(manifest_path)     # schema 校验（词表/正整数）
676         gate = quality_gate.run_gate(checked)
677         meta.extra["book118_image_pdf"] = str(image_pdf)
678         meta.extra["book118_manifest"] = str(manifest_path)
679         self._audit_row("pdf_build", doc_id, pages_measured, "ok",
680                         f"image_pdf={image_pdf.name} pages={pages_measured} "
681                         f"partial={partial} ref_chars={ref_chars}")
682         self._audit_row("quality_gate", doc_id, sum(1 for g in gate if g.passed),
683                         "ok" if all(g.passed for g in gate) else "fail",
684                         " ".join(f"{g.name}={'PASS' if g.passed else 'FAIL'}" for g in gate))
685         return PreviewPdfResult(image_pdf=image_pdf, searchable_pdf=searchable_pdf,
686                                 manifest_path=manifest_path, manifest=checked,
687                                 pages=pages_measured, page_records=records,
688                                 ref_chars=ref_chars, gate_outcomes=gate, partial=partial)
689 
690     def verify_pdf(self, result: "PreviewPdfResult") -> bool:
691         """T23 质检门收口：四门全过 且 实测页数==页清单页数≥1（partial 参照=preview_limit）。"""
692         if not result.gate_outcomes or not all(g.passed for g in result.gate_outcomes):
693             return False
694         return result.pages == len(result.page_records) and result.pages >= 1

701     def _image_transport(self, url: str) -> tuple[int, bytes]:
702         """图床 HTTP 层缺省（httpx 30s 协议直连，trust_env=False 硬口径同 _call_openapi；
703         裸 GET 无 Referer——SPEC §2/§3 探针实锤 _probe_page1.png 成功形态）。"""
704         if self._img_http is None:
705             import httpx
706             self._img_http = httpx.Client(
707                 timeout=HTTP_TIMEOUT_S, trust_env=False, follow_redirects=True,
708                 headers={"User-Agent": self._ua or _DEFAULT_UA})
709         r = self._img_http.get(url)
710         return r.status_code, r.content
