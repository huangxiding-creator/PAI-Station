## H.1 感知引擎（sense/）

```python
class SenseBus:                       # 事件总线：统一各源输出
    def publish(self, event: SenseEvent): ...   # 观察者分发给 memory/learn

class SenseEvent:
    type: str; ts: int; payload: dict
    # doc: {path, ext, size, mtime}   screen: {img_path, fg_app, fg_title}
    # im_*: {channel, chat, sender, text}   commit: {repo, sha, files}

class FSWatcher:                      # watchdog 封装
    def on_any_event(self, e): self._debounce(e)      # 3s 窗口合并
    def _flush(self, batch): SenseBus.publish(doc_batch=batch)

class ScreenLoop:                     # mss 线程
    def tick(self):
        if self._blacklisted(fg_title): return self._skip_and_log()
        img = mss.grab(virtual_screen); save_rolling(img, ttl=600)
        event = vision.describe(img) or ocr_local(img)   # 云优先本地兜底
        self._dedup_by_hash(event)                        # 相似截图（<0.1 diff）不重复上报
```
时序：watchdog 事件→去抖→MarkItDown 队列（进程池 2）→SenseEvent→memory.ingest。全链路背压：队列满时丢 screen 保 doc。

```python
def capture(ev) -> str:               # ingest 主函数
    md = to_markdown(ev)              # MarkItDown / MinerU / vision json
    facts = fast(f"{PROMPT_SUMMARIZE}\n{md[:6000]}")     # 200字摘要+实体 json_mode
    score = future_recall_prob(facts)  # 写入端审计（2.3 创新点）：
        # score = f(type权重, 实体热度, 情绪信号, 与既有记忆链接数)
    mem = Memory(...); store.insert(mem); entity_map.upsert(facts.entities)
    viking.sync(mem) if score > 0.7 else viking.index_only(mem)
    return mem.id

def recall(q, top_k=8) -> list:       # 三路召回+RRF 融合
    r1 = fts5.search(jieba(q))                       # 关键词
    r2 = entity_graph.expand(entities(q), hops=2)    # 实体图
    r3 = timeline.filter(parse_dates(q), window)     # 时间线
    fused = rrf([r1, r2, r3]); boost_by_weight(fused)
    return fused[:top_k]
```

## H.5 主动引擎（proactive/）
```python
def on_trigger(t):                    # APScheduler 回调
    task = infer_task(t)              # 系统1分类：这是什么活
    M = jtbd_score(task)              # 功能+情感+社会 × 紧迫
    A = skill_readiness(task)         # 有技能且 stable？记忆召回够？
    P = attention_ok() and not quiet_hours()
    B = M * A * P
    quadrant = (importance(task), urgency(t))
    action = dispatch(B, quadrant)    # 1.5③ 分流表
    if action in (push, drawer): executor.run_async(task, action)
    if action == push: budget.spend() # 打扰预算扣减，超支自动降 drawer
```
