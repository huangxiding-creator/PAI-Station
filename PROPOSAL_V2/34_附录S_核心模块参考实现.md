## S.1 config.py（INI 加载与校验，节选）

```python
"""配置加载器：INI -> 强类型 Config 对象，非法值中文报错。"""
from __future__ import annotations
import os
from configparser import ConfigParser
from dataclasses import dataclass
from pathlib import Path

class ConfigError(Exception):
    """配置错误，message 直接可展示给用户。"""

@dataclass(frozen=True)
class PAIConfig:
    llm: "LLMConfig"   # 其余各段同理展开

_RANGE = {"upgrade_confidence": (0.5, 0.9)}

def _get(parser: ConfigParser, section: str, key: str, default: str) -> str:
    if not parser.has_section(section):
        raise ConfigError("缺少配置段 [%s]，请对照附录 C.1 补全" % section)
    val = parser.get(section, key, fallback=default)
    if not val:
        raise ConfigError("[%s].%s 为空：该字段无默认值，必须填写" % (section, key))
    return val

def _float(section: str, key: str, raw: str) -> float:
    try:
        num = float(raw)
    except ValueError as err:
        raise ConfigError("值 '%s' 不是合法数字（[%s].%s）" % (raw, section, key)) from err
    lo, hi = _RANGE.get(key, (None, None))
    if lo is not None and not lo <= num <= hi:
        raise ConfigError("%s 超出 [%s].%s 允许范围 %s-%s" % (num, section, key, lo, hi))
    return num

def load(path="config/pai.ini") -> PAIConfig:
    file = Path(path).expanduser()
    if not file.exists():
        raise ConfigError("找不到配置文件 %s：请复制 config/pai.example.ini 为 pai.ini 后填写" % file)
    parser = ConfigParser()
    parser.read(file, encoding="utf-8")
    api_key = os.environ.get("PAI_LLM_KEY") or _get(parser, "llm", "api_key", "")
    if len(api_key) != 32:
        raise ConfigError("智谱 api_key 应为 32 位（open.bigmodel.cn 控制台复制；也可用环境变量 PAI_LLM_KEY）")
    wm = _get(parser, "channels", "wechat_mode", "vision_readonly")
    if wm != "vision_readonly":
        raise ConfigError("wechat_mode 仅允许 vision_readonly（当前 '%s'）：微信通道纯视觉只读，不可更改" % wm)
    return PAIConfig(llm=_build_llm(parser, api_key))
```

## S.2 llm/zhipu_client.py（三通道客户端，节选核心）

```python
"""智谱免费链客户端：fast/deep/vision 三通道 + 自适应限速 + 退避重试。
移植自 We-AIPO zhipu.py（已验证的 _AdaptivePacer 思想）。"""
from __future__ import annotations
import base64, json, time, urllib.request
from tenacity import retry, stop_after_attempt, wait_exponential_jitter

ENDPOINT = "https://open.bigmodel.cn/api/paas/v4/chat/completions"

class _AdaptivePacer:
    """429 时窗口翻倍、连续成功逐步恢复；全局单例，重试绝不放大并发。"""
    def __init__(self, min_interval=0.34):
        self._interval, self._min, self._last, self._ok = min_interval, min_interval, 0.0, 0
    def wait(self):
        delta = time.monotonic() - self._last
        if delta < self._interval:
            time.sleep(self._interval - delta)
        self._last = time.monotonic()
    def on_success(self):
        self._ok += 1
        if self._ok >= 20:
            self._interval = max(self._min, self._interval / 2); self._ok = 0
    def on_rate_limited(self):
        self._interval = min(8.0, self._interval * 2); self._ok = 0

class ZhipuClient:
    def __init__(self, api_key, models):
        self._key, self._models = api_key, models
        self._pacer = _AdaptivePacer()

    @retry(stop=stop_after_attempt(5), wait=wait_exponential_jitter(initial=2, max=30))
    def _post(self, payload):
        self._pacer.wait()
        req = urllib.request.Request(
            ENDPOINT, data=json.dumps(payload).encode("utf-8"),
            headers={"Authorization": "Bearer " + self._key,
                     "Content-Type": "application/json"})
        try:
            with urllib.request.urlopen(req, timeout=120) as resp:
                body = json.loads(resp.read().decode("utf-8"))
        except urllib.error.HTTPError as err:
            if err.code == 429:
                self._pacer.on_rate_limited(); raise
            detail = err.read().decode("utf-8", "ignore")[:200]
            raise RuntimeError("智谱接口错误 %s：%s" % (err.code, detail)) from err
        self._pacer.on_success()
        return body

    def fast(self, prompt, system=None, json_mode=False):
        """系统 1：摘要/分类/格式化。"""
        messages = ([{"role": "system", "content": system}] if system else []) \
                 + [{"role": "user", "content": prompt}]
        payload = {"model": self._models["fast"], "messages": messages, "temperature": 0.3}
        if json_mode:
            payload["response_format"] = {"type": "json_object"}
        body = self._post(payload)
        return {"text": body["choices"][0]["message"]["content"],
                "usage": body.get("usage", {})}

    def deep(self, prompt):
        """系统 2：显式推理链（蒸馏/根因/画像）。"""
        body = self._post({"model": self._models["deep"],
                           "messages": [{"role": "user", "content": prompt}],
                           "thinking": {"type": "enabled"}})
        msg = body["choices"][0]["message"]
        return {"text": msg["content"], "chain": msg.get("reasoning_content", ""),
                "usage": body.get("usage", {})}

    def vision(self, image_path, schema_hint):
        """截图结构化：图片 base64 + schema 约束，json 输出。"""
        with open(image_path, "rb") as fh:
            b64 = base64.b64encode(fh.read()).decode("ascii")
        body = self._post({"model": self._models["vision"],
            "messages": [{"role": "user", "content": [
                {"type": "image_url", "image_url": {"url": "data:image/jpeg;base64," + b64}},
                {"type": "text", "text": "按此 JSON schema 输出，不确定填 unknown：" + schema_hint}]}],
            "temperature": 0.1})
        return {"text": body["choices"][0]["message"]["content"]}
```

## S.3 security/audit.py（只追加审计+哈希链防篡改，全量）

```python
"""audit.db 只追加写入器：全动作留痕，哈希链防篡改。"""
import hashlib, json, sqlite3, threading, time

class Auditor:
    def __init__(self, db_path):
        self._conn = sqlite3.connect(db_path, check_same_thread=False)
        self._conn.execute("""CREATE TABLE IF NOT EXISTS audit (
            id INTEGER PRIMARY KEY AUTOINCREMENT, ts INTEGER NOT NULL,
            actor TEXT NOT NULL, action TEXT NOT NULL, target TEXT,
            detail TEXT, result TEXT NOT NULL, prev_hash TEXT)""")
        self._lock = threading.Lock()
        row = self._conn.execute(
            "SELECT prev_hash FROM audit ORDER BY id DESC LIMIT 1").fetchone()
        self._tail = row[0] if row else "GENESIS"

    def log(self, actor, action, target="", detail=None, result="ok"):
        ts = int(time.time())
        payload = json.dumps({"ts": ts, "actor": actor, "action": action,
                              "target": target, "detail": detail or {},
                              "result": result}, ensure_ascii=True, sort_keys=True)
        digest = hashlib.sha256((self._tail + payload).encode("utf-8")).hexdigest()
        with self._lock:
            self._conn.execute(
                "INSERT INTO audit(ts,actor,action,target,detail,result,prev_hash) "
                "VALUES (?,?,?,?,?,?,?)",
                (ts, actor, action, target, payload, result, digest))
            self._conn.commit()
            self._tail = digest

    def verify_chain(self):
        """完整性自检：重算哈希链，断裂即告警（附录 F Tampering 对策）。"""
        prev, ok = "GENESIS", True
        for (payload, digest) in self._conn.execute(
                "SELECT detail, prev_hash FROM audit ORDER BY id"):
            if hashlib.sha256((prev + payload).encode()).hexdigest() != digest:
                ok = False; break
            prev = digest
        return ok
```
